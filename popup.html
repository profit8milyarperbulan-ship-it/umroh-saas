async function send(action) { const [tab] = await chrome.tabs.query({ active: true, currentWindow: true }); return new Promise(resolve => { chrome.tabs.sendMessage(tab.id, { action }, res => { resolve(res); }); }); } // START document.getElementById("startBtn").onclick = async () => { await send("start"); document.getElementById("status").innerText = "Scanning..."; }; // STOP document.getElementById("stopBtn").onclick = async () => { const res = await send("stop"); document.getElementById("status").innerText = `Stopped (${res?.total || 0})`; }; // LOAD + ANALYZE async function loadData() { const res = await send("getAds"); const data = res?.data || []; document.getElementById("total").innerText = data.length; // TOP BRAND const brands = {}; data.forEach(a => { brands[a.brand] = (brands[a.brand] || 0) + 1; }); const sorted = Object.entries(brands).sort((a,b)=>b[1]-a[1]); document.getElementById("brand").innerText = sorted[0]?.[0] || "-"; // AVG COPY const avg = data.reduce((s,a)=>s+(a.body?.length||0),0)/(data.length||1); document.getElementById("avg").innerText = Math.round(avg); // LIST const list = document.getElementById("list"); list.innerHTML = ""; data.slice(0,10).forEach(ad => { const div = document.createElement("div"); div.className = "card"; div.innerHTML = ` ${ad.brand}
${ad.body} ${ad.image ? `` : ""} `; list.appendChild(div); }); } // AUTO LOAD SETIAP BUKA POPUP loadData(); // OPEN DASHBOARD document.getElementById("openDash").onclick = () => { chrome.tabs.create({ url: chrome.runtime.getURL("dashboard.html") }); };
